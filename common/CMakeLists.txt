# common/: headers (interface target siwecal_common), mappings and Python helpers; nothing to compile but tests.
install(DIRECTORY mappings/ DESTINATION ${SIWECAL_SHARE_DIR}/mappings)
# siwecal_beamspot: beam spot of a valtree and the gun tuning of the run-by-run simulation (siwecal/BeamSpot.h)
find_package(nlohmann_json 3 REQUIRED)
add_executable(siwecal_beamspot src/siwecal_beamspot.cpp)
target_link_libraries(siwecal_beamspot PRIVATE siwecal_common ROOT::Core ROOT::RIO ROOT::Tree nlohmann_json::nlohmann_json)
install(TARGETS siwecal_beamspot RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})

  target_link_libraries(test_data_path PRIVATE siwecal_common)
  add_test(NAME data_path COMMAND test_data_path ${PROJECT_SOURCE_DIR})
  add_executable(test_beam_spot tests/test_beam_spot.cpp)
  target_link_libraries(test_beam_spot PRIVATE siwecal_common)
  add_test(NAME beam_spot COMMAND test_beam_spot)
  add_test(NAME beamspot_exe COMMAND bash ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_beamspot_exe.sh $<TARGET_FILE:siwecal_beamspot>)
